cmake_minimum_required(VERSION 3.20)
project(xsai_phy_architecture LANGUAGES C)
enable_testing()
get_filename_component(XSAI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
option(XSAI_ENABLE_RVV "Compile RVV kernels (GCC >=16, RISC-V V)" OFF)
add_library(xsai_rvv_phy INTERFACE)
target_include_directories(xsai_rvv_phy INTERFACE "${XSAI_ROOT}/plugins/rvv_phy/include")
if(XSAI_ENABLE_RVV)
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "riscv" OR NOT CMAKE_C_COMPILER_ID STREQUAL "GNU" OR CMAKE_C_COMPILER_VERSION VERSION_LESS 16)
    message(FATAL_ERROR "RVV requires RISC-V and GCC >=16")
  endif()
  target_compile_options(xsai_rvv_phy INTERFACE -march=rv64gcv_zba_zbb_zbs -mabi=lp64d -mrvv-vector-bits=scalable)
else()
  target_compile_definitions(xsai_rvv_phy INTERFACE SIONNA_RVV_COMPLEX_DISABLE=1)
endif()
add_library(xsai_oai_adapter INTERFACE)
target_link_libraries(xsai_oai_adapter INTERFACE xsai_rvv_phy)
target_include_directories(xsai_oai_adapter INTERFACE "${XSAI_ROOT}/adapters/oai/include")
add_executable(test_phy_adapter tests/test_phy_adapter.c)
target_link_libraries(test_phy_adapter PRIVATE xsai_oai_adapter)
target_compile_options(test_phy_adapter PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(test_phy_adapter PRIVATE EXPECT_RVV=$<BOOL:${XSAI_ENABLE_RVV}>)
add_test(NAME phy_adapter_contract COMMAND test_phy_adapter)
